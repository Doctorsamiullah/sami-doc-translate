### 1. Translation Engine Contract

Both the Windows (Python) and Android (Kotlin) applications adhere to identical interface contracts for translation:

#### Python Interface (`windows-app/src/core/interfaces.py`):
```python
from abc import ABC, abstractmethod
from typing import List, Callable, Optional

class ITranslationEngine(ABC):
    @abstractmethod
    def translate(self, text: str, src: str, tgt: str) -> str:
        """Translates a single string between 'en' and 'hi'."""
        pass

    @abstractmethod
    def translate_batch(self, texts: List[str], src: str, tgt: str) -> List[str]:
        """Translates a batch of strings concurrently/in parallel."""
        pass

    @abstractmethod
    def is_ready(self, src: str, tgt: str) -> bool:
        """Checks if local model weights are downloaded and initialized."""
        pass

    @abstractmethod
    def download_models(self, progress_callback: Callable[[float, str], None]) -> bool:
        """Downloads offline model packs with progress callbacks."""
        pass
```

#### Kotlin Interface (`android-app/app/.../domain/ITranslator.kt`):
```kotlin
interface ITranslationEngine {
    suspend fun translate(text: String, src: String, tgt: String): Result<String>
    suspend fun translateBatch(texts: List<String>, src: String, tgt: String): Result<List<String>>
    suspend fun isModelAvailable(lang: String): Boolean
    suspend fun downloadModel(lang: String, onProgress: (Float) -> Unit): Result<Unit>
}
```

### 2. Document Streaming Parser & Writer Contracts

To support unlimited page counts without memory spikes, parsers emit runs lazily page-by-page:

```python
class IDocumentStream(ABC):
    @abstractmethod
    def page_count(self) -> int:
        pass

    @abstractmethod
    def stream_pages(self) -> Generator[DocumentPage, None, None]:
        """Yields one page of text runs, tables, and images at a time."""
        pass

class IDocumentWriter(ABC):
    @abstractmethod
    def open(self, target_path: str) -> None:
        pass

    @abstractmethod
    def write_page(self, page: DocumentPage) -> None:
        """Appends a translated page to disk output stream."""
        pass

    @abstractmethod
    def close(self) -> None:
        pass
```

### 3. Format Converter Protocol

```python
class IFormatConverter(ABC):
    @abstractmethod
    def can_convert(self, src_ext: str, tgt_ext: str) -> bool:
        """Returns True if direct format conversion is supported."""
        pass

    @abstractmethod
    def convert_file(self, src_path: str, tgt_path: str, progress_callback: Optional[Callable[[int, int], None]] = None) -> bool:
        """Performs format conversion without translation."""
        pass


class IFileCompressor(ABC):
    @abstractmethod
    def compress(
        self,
        src_path: str,
        tgt_path: str,
        target_size_bytes: int,
        preserve_readability: bool = True,
        progress_callback: Optional[Callable[[float, str], None]] = None
    ) -> bool:
        """
        Compresses document to target size while preserving readability.
        Guarantees scratch buffer cleanup upon completion.
        """
        pass

    @abstractmethod
    def cleanup_scratch_buffers(self) -> None:
        """Wipes all temporary compressor cache and scratch files."""
        pass
```
